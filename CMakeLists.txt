cmake_minimum_required(VERSION 3.20)
project(multi_copyname VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(names_tests tests/names_tests.cpp)
  target_include_directories(names_tests PRIVATE src)
  if(MSVC)
    target_compile_options(names_tests PRIVATE /utf-8)
  endif()
  add_test(NAME names COMMAND names_tests)
endif()
if(WIN32)
  enable_language(RC)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This release requires Windows x64")
  endif()
  add_library(multi_copyname_shell SHARED src/shell.cpp src/shell.def)
  set_target_properties(multi_copyname_shell PROPERTIES PREFIX "")
  add_executable(multi_copyname WIN32 src/app.cpp src/app.rc)
  target_include_directories(multi_copyname PRIVATE src)
  if(MSVC)
    # app.rc already embeds the complete manifest; do not emit a second one.
    target_link_options(multi_copyname PRIVATE /MANIFEST:NO)
  endif()
  foreach(target multi_copyname multi_copyname_shell)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
    target_link_libraries(${target} PRIVATE ole32 shell32 user32 advapi32 uuid)
    if(MSVC)
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /EHsc)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endforeach()
  target_link_libraries(multi_copyname PRIVATE gdi32 comctl32)
  if(MINGW)
    target_link_options(multi_copyname PRIVATE -municode)
  endif()
  install(TARGETS multi_copyname multi_copyname_shell RUNTIME DESTINATION . LIBRARY DESTINATION .)
  if(BUILD_TESTING)
    add_executable(shell_tests tests/shell_tests.cpp)
    target_include_directories(shell_tests PRIVATE src)
    target_compile_definitions(shell_tests PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0602)
    target_link_libraries(shell_tests PRIVATE ole32 shell32 user32 uuid)
    if(MSVC)
      target_compile_options(shell_tests PRIVATE /utf-8)
    endif()
    add_dependencies(shell_tests multi_copyname_shell)
    add_test(NAME shell_integration COMMAND shell_tests)
    if(MINGW)
      target_link_options(names_tests PRIVATE -static -static-libgcc -static-libstdc++)
      target_link_options(shell_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
endif()
